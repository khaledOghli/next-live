<DemoSection
  title="Hooks and line numbers"
  description={
    <>
      Fence metadata like <code>title=&quot;Counter&quot;</code> and{' '}
      <code>showLineNumbers</code> forward to the editor, same as Docusaurus static blocks.
    </>
  }
>

```tsx live title="Counter" showLineNumbers
import { useState } from 'react';
import { Button } from '@next-live-docusaurus/modules';

export default function Counter() {
  const [n, setN] = useState(0);
  return (
    <Button onClick={() => setN((v) => v + 1)}>
      Count: {n}
    </Button>
  );
}
```
